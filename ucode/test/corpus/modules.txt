================================================================================

--------------------------------------------------------------------------------

; statement form `import "mod.uc"` is supported. The `import` keyword
; produces an ERROR node; the parenthesised path is recovered as the
; variable's value.

  (lexical_declaration
    (variable_declarator

================================================================================
Empty export list is rejected — produces ERROR
================================================================================

export {};

--------------------------------------------------------------------------------

; ucode's export-list parser demands a label (or `default`) before it will
; even check for '}' — an empty `{}` is "Expecting Label", so recovery leaves
; a MISSING identifier where the first specifier's name should be.

(program
  (export_statement
    clause: (export_clause
      (export_specifier
        name: (MISSING identifier)))))
